class Transaction

        cin >> type;

        cin >> amount;

};
bool idExists(int id)
{
    ifstream file("bank.dat", ios::binary);
    Transaction t;

    while (file.read((char*)&t, sizeof(t)))
    {
        if (t.id == id)
            return true;
    }

    return false;
}

void addIndex(int roll, long pos)
{
    ofstream index("index.dat", ios::binary | ios::app);

    index.write((char*)&roll, sizeof(roll));
    index.write((char*)&pos, sizeof(pos));

    index.close();
}

long findPosition(int roll)
{
    ifstream index("index.dat", ios::binary);

    int r;
    long pos;

    while (index.read((char*)&r, sizeof(r)))
    {
        index.read((char*)&pos, sizeof(pos));

        if (r == roll)
        {
            index.close();
            return pos;
        }
    }

    index.close();
    return -1;
}

void insertStudent()
{
    Student s;
    s.input();

    if (findPosition(s.rollNo) != -1)
    {
        cout << "Roll number already exists.\n";
        return;
    }

    fstream file("students.dat",
                 ios::binary | ios::in | ios::out);

    if (!file)
    {

        ofstream create("students.dat", ios::binary);
        create.close();

        file.open("students.dat",
                  ios::binary | ios::in | ios::out);
    }

    Student temp;
    long pos = -1;

    while (file.read((char*)&temp, sizeof(temp)))
    {
        if (!temp.active)
        {
            pos = file.tellg() - sizeof(temp);
            break;
        }
    }

    if (pos == -1)
    {
        file.clear();
        file.seekp(0, ios::end);
        pos = file.tellp();
    }

    file.seekp(pos);
    file.write((char*)&s, sizeof(s));

    file.close();

    addIndex(s.rollNo, pos);

    cout << "Student inserted successfully.\n";
}

void sortByCGPA()
{
    Student arr[100];
    int n = 0;

    ifstream file("students.dat", ios::binary);

    while (file.read((char*)&arr[n], sizeof(Student)))
    {
        if (arr[n].active)
            n++;
    }

    file.close();

    // Bubble Sort
    for (int i = 0; i < n - 1; i++)
    {
        for (int j = 0; j < n - i - 1; j++)
        {
            if (arr[j].cgpa < arr[j + 1].cgpa)
            {
                Student temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }

    cout << "\nStudents sorted by CGPA:\n";
    cout << "Roll\tName\tDept\tSem\tCGPA\n";

    for (int i = 0; i < n; i++)
        arr[i].display();
}

void updateBill()

    while (file.read((char*)&p, sizeof(p)))
    {
        if (p.patientID == id)

            double oldBill = p.bill;

            cout << "Old Bill: " << oldBill << endl;

            cout << "Enter New Bill: ";

            cout << "Bill updated successfully.\n";
            break;
        }
    }

    if (!found)
        cout << "Patient not found.\n";

    file.close();
}
